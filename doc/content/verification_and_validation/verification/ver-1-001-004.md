## Problem Statement

!style halign=left
xxx

\begin{equation} \label{eq:standard-diffusion}
 \text{sign} \mu \vec{E} n - \nabla \cdot \left(D \nabla n \right) = S_{n}
\end{equation}

\begin{equation} \label{eq:standard-diffusion}
 -\nabla \cdot \left( \varepsilon \nabla V \right) =  \e \left( n_{i} - n_{e} \right)+ S_{n}
\end{equation}

XXX

\begin{equation}
\frac{\partial n_{\epsilon}}{\partial t} + \nabla \cdot \vec{\Gamma_{\epsilon}} - \vec{E} \cdot \vec{\Gamma_{e}} = S_{\epsilon} \\[10pt]
\vec{\Gamma}_{\epsilon} = - \mu_{\epsilon} \vec{E} n_{\epsilon} - D_{\epsilon} \nabla n_{\epsilon}
\end{equation}

XXX

\begin{equation} \label{eq:logarithmic-diffusion}
  n = \text{exp} N
\end{equation}

\begin{equation} \label{eq:logarithmic-diffusion}
  \text{sign} \mu \vec{E} \text{exp} \left( N \right) - \nabla \cdot \left(D \text{exp} \left( N \right) \nabla N \right) = S_{n}
\end{equation}

\begin{equation} \label{eq:standard-diffusion}
 -\nabla \cdot \left( \varepsilon \nabla V \right) =  \e \left( \text{exp} \left( N_{i} \right) - \text{exp} \left( N_{e} \right) \right)+ S_{n}
\end{equation}

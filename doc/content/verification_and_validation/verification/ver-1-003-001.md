\begin{equation} \label{eq:standard-diffusion}
 \text{sign} \mu \vec{E} n - \nabla \cdot \left(D \nabla n \right) = S_{n}
\end{equation}

\begin{equation} \label{eq:standard-diffusion}
  \vec{E} = \frac{D_{i} - D_{e}}{\mu_{i} - \mu_{e}} \frac{\nabla n}{n}
\end{equation}

\begin{equation} \label{eq:standard-diffusion}
  \vec{\Gamma} = D_{a} \nabla n
\end{equation}

\begin{equation} \label{eq:standard-diffusion}
  D_{a} = \frac{\mu_{i} D_{e} + \mu_{e} D_{i}}{\mu_{i} + \mu_{e}}
\end{equation}

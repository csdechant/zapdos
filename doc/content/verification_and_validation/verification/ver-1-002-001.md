\begin{equation} \label{eq:standard-diffusion}
  \nabla^{2} = \frac{e}{\varepsilon_{0}} \left(n_{e} - n_{i} \right)
\end{equation}

\begin{equation} \label{eq:standard-diffusion}
  n_{e} = n_{0} e^{V/T_{e}}
\end{equation}
\begin{equation} \label{eq:standard-diffusion}
  n_{i} = n_{0}
\end{equation}

\begin{equation} \label{eq:standard-diffusion}
  \nabla^{2} V = \frac{e n_{0}}{\varepsilon_{0}} \left( e^{V/T_{e}} - 1 \right)
\end{equation}
